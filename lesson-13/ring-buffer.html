発展編リングバッファ

末尾の次は、先頭へ。キューの空いた場所を繰り返し使います。

ex13-7.c

右の図を操作する →
#include <stdio.h>

#define QUEUE_SIZE 5

int queue[QUEUE_SIZE];
int *front = queue;
int *rear = queue;
int count = 0;

/* 次の要素を指すポインタを返す */
int *next_pointer(int *p) {
    p++;
    if (p == queue + QUEUE_SIZE) {
        p = queue;
    }
    return p;
}

int enqueue(int x) {
    if (count == QUEUE_SIZE) {
        return 0;  /* 満杯 */
    }
    *rear = x;
    rear = next_pointer(rear);
    count++;
    printf("IN : %d\n", x);
    return 1;
}

int dequeue(int *x) {
    if (count == 0) {
        return 0;  /* 空 */
    }
    *x = *front;
    front = next_pointer(front);
    count--;
    printf("OUT: %d\n", *x);
    return 1;
}

int main(void) {
    int x;

    enqueue(10);
    enqueue(20);
    enqueue(30);
    enqueue(40);
    enqueue(50);

    if (!enqueue(60)) {
        printf("キューが満杯です．\n");
    }

    dequeue(&x);  /* 10 */
    dequeue(&x);  /* 20 */

    /* 空いた先頭側の領域が再利用される */
    enqueue(60);
    enqueue(70);

    while (dequeue(&x)) {
    }

    return 0;
}

容量は5要素。front・rear の折り返しと、count による空・満杯の判定に注目します。

−99〜99 の整数。満杯なら追加できず、空なら取り出せません。

図の見方

配列のメモリは連続しています。円は「末尾の次に先頭へ戻る」動きの表現です。薄い値は使用していない場所に残った値です。

アドレスは説明用。int を4バイトとして描いています。
